   179→/** Keep only nav items the user may view; drop groups that become empty. */
   180→function filterNav(user: SessionUser): NavGroup[] {
   181→  return NAV
   182→    .map((g) => ({
   183→      ...g,
   184→      items: g.items.filter((item) => {
   185→        const moduleName = NAV_PERMISSIONS[item.key]
   186→        return !moduleName || canView(user.permissions, moduleName)
   187→      }),
   188→    }))
   189→    .filter((g) => g.items.length > 0)
   190→}

   192→function userInitials(name: string): string {
   193→  const parts = name.trim().split(/\s+/).slice(0, 2)
   194→  const chars = parts.map((p) => p[0]?.toUpperCase() ?? '').join('')
   195→  return chars || 'U'
   196→}

   198→function useHashRoute(allowed: Set<string>): string {
   199→  const [route, setRoute] = useState('dashboard')
   200→  useEffect(() => {
   201→    const read = () => {
   202→      const h = window.location.hash.replace(/^#\/?/, '') || 'dashboard'
   203→      // Hidden (permission-gated) or unknown modules fall back to the dashboard
   204→      setRoute(h in MODULE_VIEWS && allowed.has(h) ? h : 'dashboard')
   205→    }
   206→    read()
   207→    window.addEventListener('hashchange', read)
   208→    return () => window.removeEventListener('hashchange', read)
   209→  }, [allowed])
   210→  return route
   211→}

   231→/** localStorage key persisting which sidebar groups are expanded */
   232→const NAV_OPEN_KEY = 'dolinext_nav_open'
   233→
   234→function readOpenGroups(): Set<string> {
   235→  try {
   236→    const raw = window.localStorage.getItem(NAV_OPEN_KEY)
   237→    if (raw) return new Set(JSON.parse(raw) as string[])
   238→  } catch {
   239→    // storage blocked — fall through to empty set
   240→  }
   241→  return new Set()
   242→}
   243→
   244→function persistOpenGroups(groups: Set<string>) {
   245→  try {
   246→    window.localStorage.setItem(NAV_OPEN_KEY, JSON.stringify([...groups]))
   247→  } catch {
   248→    // storage blocked — state still lives for this mount
   249→  }
   250→}

   256→/**
   257→ * Sidebar menu: parent groups (collapsible) → child module links.
   258→ * Open/closed state persists in localStorage; the group holding the active
   259→ * route is always expanded, and its parent shows a subtle active hint while
   260→ * collapsed so users never lose their place.
   261→ */
   262→function SidebarContent({ current, nav }: { current: string; nav: NavGroup[] }) {
   263→  const [openGroups, setOpenGroups] = useState<Set<string>>(() => {
   264→    const stored = readOpenGroups()
   265→    if (stored.size > 0) return stored
   266→    // First visit: expand the active group (falls back to Home).
   267→    const activeGroup = nav.find((g) => g.items.some((i) => i.key === current))?.group
   268→    return new Set([activeGroup ?? 'Home'])
   269→  })
   270→
   271→  // Keep the active route's group expanded (also covers search/quick-create jumps).
   272→  // Derived-state adjustment during render — the React-recommended pattern.
   273→  const [prevCurrent, setPrevCurrent] = useState(current)
   274→  if (prevCurrent !== current) {
   275→    setPrevCurrent(current)
   276→    const group = nav.find((g) => g.items.some((i) => i.key === current))?.group
   277→    if (group && !openGroups.has(group)) {
   278→      const next = new Set(openGroups)
   279→      next.add(group)
   280→      persistOpenGroups(next)
   281→      setOpenGroups(next)
   282→    }
   283→  }
   284→
   285→  function toggleGroup(group: string) {
   286→    setOpenGroups((prev) => {
   287→      const next = new Set(prev)
   288→      if (next.has(group)) next.delete(group)
   289→      else next.add(group)
   290→      persistOpenGroups(next)
   291→      return next
   292→    })
   293→  }
   294→
   295→  return (
   296→    <div className="flex h-full flex-col">
   297→      <div className="px-4 py-4">
   298→        <Logo />
   299→      </div>
   300→      <ScrollArea className="flex-1 px-2 pb-4">
   301→        <nav aria-label="ERP modules" className="space-y-0.5">
   302→          {nav.map((g) => {
   303→            const open = openGroups.has(g.group)
   304→            const activeChild = g.items.some((i) => i.key === current)
   305→            return (
   306→              <div key={g.group}>
   307→                <button
   308→                  type="button"
   309→                  onClick={() => toggleGroup(g.group)}
   310→                  aria-expanded={open}
   311→                  aria-controls={groupSlug(g.group)}
   312→                  className={cn(
   313→                    'mt-1 flex w-full items-center gap-2.5 rounded-md px-2.5 py-2 text-sm font-medium transition-colors hover:bg-accent focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring',
   314→                    activeChild && !open
   315→                      ? 'text-foreground'
   316→                      : 'text-foreground/75',
   317→                  )}
   318→                >
   319→                  <g.icon
   320→                    className={cn(
   321→                      'h-4 w-4 shrink-0',
   322→                      activeChild ? 'text-primary' : 'text-muted-foreground',
   323→                    )}
   324→                    aria-hidden="true"
   325→                  />
   326→                  <span className="flex-1 truncate text-left">{g.group}</span>
   327→                  <ChevronDown
   328→                    className={cn(
   329→                      'h-4 w-4 shrink-0 text-muted-foreground transition-transform duration-200',
   330→                      open && 'rotate-180',
   331→                    )}
   332→                    aria-hidden="true"
   333→                  />
   334→                </button>
   335→                {open && (
   336→                  <ul id={groupSlug(g.group)} className="ml-[26px] mt-0.5 space-y-0.5 border-l pl-2">
   337→                    {g.items.map((item) => {
   338→                      const active = current === item.key
   339→                      return (
   340→                        <li key={item.key}>
   341→                          <button
   342→                            type="button"
   343→                            onClick={() => navigate(item.key)}
   344→                            aria-current={active ? 'page' : undefined}
   345→                            className={cn(
   346→                              'flex w-full items-center gap-2.5 rounded-md px-2.5 py-2 text-sm transition-colors',
   347→                              active
   348→                                ? 'bg-primary font-medium text-primary-foreground shadow-sm'
   349→                                : 'text-foreground/80 hover:bg-accent hover:text-foreground',
   350→                            )}
   351→                          >
   352→                            <item.icon className="h-4 w-4 shrink-0" aria-hidden="true" />
   353→                            <span className="truncate">{item.label}</span>
   354→                          </button>
   355→                        </li>
   356→                      )
   357→                    })}
   358→                  </ul>
   359→                )}
   360→              </div>
   361→            )
   362→          })}
   363→        </nav>
   364→      </ScrollArea>
   365→      <div className="border-t px-4 py-3 text-[11px] text-muted-foreground">
   366→        Rebuilt from dolibarr.org · v1.0
   367→      </div>
   368→    </div>
   369→  )
   370→}

   446→function TopBar({

   610→/**
   611→ * Default export — the auth-aware root.
   612→ * loading → splash · anon → AuthScreen · authed → ErpApp (fresh per user).
   613→ */
   614→export default function ErpRoot() {
   615→  const status = useAuthStore((s) => s.status)
   616→  const user = useAuthStore((s) => s.user)
   617→  const init = useAuthStore((s) => s.init)
   618→  const forceSignOut = useAuthStore((s) => s.forceSignOut)
   619→
   620→  // Restore the session once on mount
   621→  useEffect(() => {
   622→    void init()
   623→  }, [init])
   624→
   625→  // Session watchdog: immediate check + every 60s + on focus/visibility.
   626→  // Silent while authed — only a 401 drops the local session.
   627→  useEffect(() => {
   628→    if (status !== 'authed') return
   629→    let alive = true
   630→    const check = async () => {
   631→      try {
   632→        const res = await fetch('/api/auth/me', {
   633→          credentials: 'same-origin',
   634→          cache: 'no-store',
   635→          headers: authHeaders(),
   636→        })
   637→        if (alive && res.status === 401) {
   638→          forceSignOut()
   639→          window.location.hash = '#/dashboard'
   640→        }
   641→      } catch {
   642→        // network hiccup — keep the current session state
   643→      }
   644→    }
   645→    void check()
   646→    const id = window.setInterval(() => { void check() }, 60_000)
   647→    const onVisible = () => {
   648→      if (document.visibilityState === 'visible') void check()
   649→    }
   650→    window.addEventListener('focus', onVisible)
   651→    document.addEventListener('visibilitychange', onVisible)
   652→    return () => {
   653→      alive = false
   654→      window.clearInterval(id)
   655→      window.removeEventListener('focus', onVisible)
   656→      document.removeEventListener('visibilitychange', onVisible)
   657→    }
   658→  }, [status, forceSignOut])
   659→
   660→  if (status === 'loading') return <Splash />
   661→  if (status === 'anon' || !user) return <AuthScreen />
   662→  return <ErpApp key={user.id} user={user} />
   663→}

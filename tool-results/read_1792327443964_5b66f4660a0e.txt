    42→export default function CoffeeModule() {
    43→  const user = useAuthStore((s) => s.user)
    44→  const mayEdit = user ? canEdit(user.permissions, 'coffee') : false
    45→  // Super administrator only — the KHQR image must belong to the account owner.
    46→  const mayUpload = user ? isSuperAdmin(user.permissions) : false
    47→  const [tab, setTab] = useState('page')
    48→  const queryClient = useQueryClient()
    49→
    50→  const q = useQuery({ queryKey: ['coffee'], queryFn: fetchCoffeePage })
    51→
    52→  // Shared config saver — writes the page config and refreshes the cache.
    53→  const [savingPage, setSavingPage] = useState(false)
    54→  async function savePage(patch: Record<string, unknown>, successMsg: string): Promise<boolean> {
    55→    setSavingPage(true)
    56→    try {
    57→      const page = await putCoffeePage(patch)
    58→      queryClient.setQueryData<CoffeePayload>(['coffee'], (old) =>
    59→        old ? { ...old, page } : old,
    60→      )
    61→      toast.success(successMsg)
    62→      return true
    63→    } catch (err) {
    64→      toast.error(err instanceof Error ? err.message : 'Failed to save the coffee page')
    65→      return false
    66→    } finally {
    67→      setSavingPage(false)
    68→    }
    69→  }
    70→
    71→  // KHQR image upload / removal — saved immediately (server config is the base).
    72→  const [uploading, setUploading] = useState(false)
    73→  async function handleImageFile(file: File) {
    74→    if (!/^image\/(png|jpe?g|webp)$/.test(file.type)) {
    75→      toast.error('Only PNG, JPG or WebP images are supported.')
    76→      return
    77→    }
    78→    if (file.size > MAX_IMAGE_BYTES) {
    79→      toast.error('Image is too large — the maximum size is 2 MB.')
    80→      return
    81→    }
    82→    setUploading(true)
    83→    try {
    84→      const dataUrl = await readFileAsDataUrl(file)
    85→      await savePage({ khqrImage: dataUrl }, 'KHQR image uploaded')
    86→    } catch {
    87→      toast.error('Could not read that file — please try another image.')
    88→    } finally {
    89→      setUploading(false)
    90→    }
    91→  }
    92→  async function removeImage() {
    93→    setUploading(true)
    94→    try {
    95→      await savePage({ khqrImage: null }, 'KHQR image removed')
    96→    } finally {
    97→      setUploading(false)
    98→    }
    99→  }
   100→
   101→  // Support log mutations
   102→  const [deletingId, setDeletingId] = useState<string | null>(null)
   103→  async function deleteSupport(id: string) {
   104→    setDeletingId(id)
   105→    try {
   106→      await api(`/api/coffee/supports/${id}`, { method: 'DELETE' })
   107→      toast.success('Support entry removed')
   108→      await queryClient.invalidateQueries({ queryKey: ['coffee'] })
   109→    } catch (err) {
   110→      toast.error(err instanceof Error ? err.message : 'Failed to delete the support entry')
   111→    } finally {
   112→      setDeletingId(null)
   113→    }
   114→  }
   115→  async function addSupport(body: {
   116→    name: string
   117→    amount: number
   118→    currency: Currency
   119→    message: string | null
   120→    date: string
   121→  }): Promise<boolean> {
   122→    try {
   123→      await api('/api/coffee/supports', { method: 'POST', body: JSON.stringify(body) })
   124→      toast.success('Support recorded — it now shows on your page')
   125→      await queryClient.invalidateQueries({ queryKey: ['coffee'] })
   126→      return true
   127→    } catch (err) {
   128→      toast.error(err instanceof Error ? err.message : 'Failed to record the support')
   129→      return false
   130→    }
   131→  }
   132→
   133→  const page = q.data?.page
   134→  const supporters = q.data?.supporters ?? []
   135→
   136→  return (
   137→    <div className="space-y-6">
   138→      <PageHeader
   139→        title="Buy Me a Coffee"
   140→        description="A friendly supporter page with your KHQR payment QR — scan to send a coffee."
   141→        icon={Coffee}
   142→        actions={
   143→          page ? (
   144→            page.khqrImage ? (
   145→              <Badge variant="outline" className="gap-1.5 border-emerald-600/30 text-emerald-700 dark:text-emerald-400">
   146→                <QrCode className="h-3.5 w-3.5" aria-hidden="true" />
   147→                KHQR ready
   148→              </Badge>
   149→            ) : (
   150→              <Badge variant="outline" className="gap-1.5 border-amber-600/40 text-amber-700 dark:text-amber-400">
   151→                <QrCode className="h-3.5 w-3.5" aria-hidden="true" />
   152→                QR not set yet
   153→              </Badge>
   154→            )
   155→          ) : undefined
   156→        }
   157→      />
   158→
   159→      {q.isLoading && (
   160→        <div className="space-y-4">
   161→          <Skeleton className="h-64 w-full rounded-xl" />
   162→          <div className="grid gap-4 sm:grid-cols-2">
   163→            <Skeleton className="h-40 rounded-xl" />
   164→            <Skeleton className="h-40 rounded-xl" />
   165→          </div>
   166→        </div>
   167→      )}
   168→
   169→      {q.isError && (
   170→        <Alert variant="destructive">
   171→          <Coffee className="h-4 w-4" aria-hidden="true" />
   172→          <AlertTitle>Failed to load the coffee page</AlertTitle>
   173→          <AlertDescription>{(q.error as Error).message}</AlertDescription>
   174→          <Button variant="outline" size="sm" className="mt-3 w-fit" onClick={() => void q.refetch()}>
   175→            Retry
   176→          </Button>
   177→        </Alert>
   178→      )}
   179→
   180→      {q.isSuccess && page && (
   181→        <Tabs value={tab} onValueChange={setTab}>
   182→          <TabsList className="w-full justify-start overflow-x-auto sm:w-auto">
   183→            <TabsTrigger value="page" className="gap-1.5">
   184→              <Coffee className="h-4 w-4" aria-hidden="true" />
   185→              Coffee page
   186→            </TabsTrigger>
   187→            <TabsTrigger value="settings" className="gap-1.5">
   188→              <ImagePlus className="h-4 w-4" aria-hidden="true" />
   189→              Settings
   190→            </TabsTrigger>
   191→            <TabsTrigger value="log" className="gap-1.5">
   192→              <HeartHandshake className="h-4 w-4" aria-hidden="true" />
   193→              Support log
   194→            </TabsTrigger>
   195→          </TabsList>
   196→
   197→          <TabsContent value="page" className="mt-4">
   198→            <CoffeePageView
   199→              key={page.updatedAt ?? 'default'}
   200→              page={page}
   201→              supporters={supporters}
   202→              mayUpload={mayUpload}
   203→              onGoSettings={() => setTab('settings')}
   204→            />
   205→          </TabsContent>
   206→
   207→          <TabsContent value="settings" className="mt-4">
   208→            <CoffeeSettingsTab
   209→              key={page.updatedAt ?? 'default'}
   210→              page={page}
   211→              savingPage={savingPage}
   212→              uploading={uploading}
   213→              mayEdit={mayEdit}
   214→              mayUpload={mayUpload}
   215→              onImageFile={(f) => void handleImageFile(f)}
   216→              onRemoveImage={() => void removeImage()}
   217→              onSavePatch={(patch, msg) => savePage(patch, msg)}
   218→            />
   219→          </TabsContent>
   220→
   221→          <TabsContent value="log" className="mt-4">
   222→            <SupportLogTab
   223→              page={page}
   224→              supporters={supporters}
   225→              mayEdit={mayEdit}
   226→              onAdd={addSupport}
   227→              onDelete={deleteSupport}
   228→              deletingId={deletingId}
   229→            />
   230→          </TabsContent>
   231→        </Tabs>
   232→      )}
   233→    </div>
   234→  )
   235→}

   477→// ---------------------------------------------------------------------------
   478→// Settings tab — KHQR upload + page details
   479→// ---------------------------------------------------------------------------
   480→function CoffeeSettingsTab({
   481→  page, savingPage, uploading, mayEdit, mayUpload, onImageFile, onRemoveImage, onSavePatch,
   482→}: {
   483→  page: CoffeePageCfg
   484→  savingPage: boolean
   485→  uploading: boolean
   486→  mayEdit: boolean
   487→  mayUpload: boolean
   488→  onImageFile: (file: File) => void
   489→  onRemoveImage: () => void
   490→  onSavePatch: (patch: Record<string, unknown>, successMsg: string) => Promise<boolean>
   491→}) {
   492→  const [displayName, setDisplayName] = useState(page.displayName)
   493→  const [tagline, setTagline] = useState(page.tagline)
   494→  const [currency, setCurrency] = useState<Currency>(page.currency)
   495→  const [presetsInput, setPresetsInput] = useState(page.presetAmounts)
   496→  const [khqrName, setKhqrName] = useState(page.khqrName ?? '')
   497→  const [khqrBank, setKhqrBank] = useState(page.khqrBank ?? '')
   498→
   499→  async function handleSave() {
   500→    const name = displayName.trim()
   501→    if (!name) { toast.error('Display name is required'); return }
   502→    if (tagline.trim().length > 200) { toast.error('Keep the tagline under 200 characters'); return }
   503→
   504→    const parts = presetsInput.split(',').map((p) => p.trim()).filter(Boolean)
   505→    if (parts.length === 0) { toast.error('Add at least one preset amount, e.g. 1,3,5,10'); return }
   506→    if (parts.length > 8) { toast.error('Use at most 8 preset amounts'); return }
   507→    for (const p of parts) {
   508→      const n = Number(p)
   509→      if (!Number.isFinite(n) || n <= 0) { toast.error(`"${p}" is not a valid amount`); return }
   510→    }
   511→
   512→    const ok = await onSavePatch(
   513→      {
   514→        displayName: name,
   515→        tagline: tagline.trim(),
   516→        currency,
   517→        presetAmounts: parts.join(','),
   518→        khqrName: khqrName.trim() || null,
   519→        khqrBank: khqrBank.trim() || null,
   520→      },
   521→      'Coffee page saved',
   522→    )
   523→    if (ok) setPresetsInput(parts.join(','))
   524→  }
   525→
   526→  return (

   851→/** Keyed conditional mount (fresh state per open) — add a support log entry. */
   852→function AddSupportDialog({
   853→  defaultCurrency, onClose, onSubmit,
   854→}: {
   855→  defaultCurrency: Currency
   856→  onClose: () => void
   857→  onSubmit: (body: { name: string; amount: number; currency: Currency; message: string | null; date: string }) => Promise<void>
   858→}) {
   859→  const [name, setName] = useState('Anonymous')
   860→  const [amount, setAmount] = useState('')
   861→  const [currency, setCurrency] = useState<Currency>(defaultCurrency)
   862→  const [message, setMessage] = useState('')

   864→  const [saving, setSaving] = useState(false)
   865→
   866→  async function handleSubmit() {
   867→    const n = Number(amount)
   868→    if (!name.trim()) { toast.error('Name is required'); return }
   869→    if (!Number.isFinite(n) || n <= 0) { toast.error('Enter an amount greater than zero'); return }
   870→    if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) { toast.error('Pick a valid date'); return }
   871→    setSaving(true)

   873→      await onSubmit({
   874→        name: name.trim(),
   875→        amount: n,
   876→        currency,
   877→        message: message.trim() || null,
   878→        date,
   879→      })
   880→    } finally {
   881→      setSaving(false)
   882→    }
   883→  }
   884→
   885→  return (
   886→    <Dialog open onOpenChange={(open) => { if (!open) onClose() }}>

   899→                id="support-name"
   900→                value={name}
   901→                maxLength={60}
   902→                onChange={(e) => setName(e.target.value)}
   903→                placeholder="Anonymous"

   905→            </div>
   906→            <div className="space-y-1.5">
   907→              <Label htmlFor="support-date">Date</Label>

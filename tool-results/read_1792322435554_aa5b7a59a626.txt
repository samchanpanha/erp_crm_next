   192→   192→function useHashRoute(allowed: Set<string>): string {
   193→   193→  const [route, setRoute] = useState('dashboard')
   194→   194→  useEffect(() => {
   195→   195→    const read = () => {
   196→   196→      const h = window.location.hash.replace(/^#\/?/, '') || 'dashboard'
   197→   197→      // Hidden (permission-gated) or unknown modules fall back to the dashboard
   198→   198→      setRoute(h in MODULE_VIEWS && allowed.has(h) ? h : 'dashboard')
   199→   199→    }
   200→   200→    read()
   201→   201→    window.addEventListener('hashchange', read)
   202→   202→    return () => window.removeEventListener('hashchange', read)
   203→   203→  }, [allowed])
   204→   204→  return route
   205→   205→}

   225→   225→/** localStorage key persisting which sidebar groups are expanded */
   226→   226→const NAV_OPEN_KEY = 'dolinext_nav_open'
   227→   227→
   228→   228→function readOpenGroups(): Set<string> {
   229→   229→  try {
   230→   230→    const raw = window.localStorage.getItem(NAV_OPEN_KEY)
   231→   231→    if (raw) return new Set(JSON.parse(raw) as string[])
   232→   232→  } catch {
   233→   233→    // storage blocked — fall through to empty set
   234→   234→  }
   235→   235→  return new Set()
   236→   236→}
   237→   237→
   238→   238→function persistOpenGroups(groups: Set<string>) {
   239→   239→  try {
   240→   240→    window.localStorage.setItem(NAV_OPEN_KEY, JSON.stringify([...groups]))
   241→   241→  } catch {
   242→   242→    // storage blocked — state still lives for this mount
   243→   243→  }
   244→   244→}

   250→   250→/**
   251→   251→ * Sidebar menu: parent groups (collapsible) → child module links.
   252→   252→ * Open/closed state persists in localStorage; the group holding the active
   253→   253→ * route is always expanded, and its parent shows a subtle active hint while
   254→   254→ * collapsed so users never lose their place.
   255→   255→ */
   256→   256→function SidebarContent({ current, nav }: { current: string; nav: NavGroup[] }) {
   257→   257→  const [openGroups, setOpenGroups] = useState<Set<string>>(() => {
   258→   258→    const stored = readOpenGroups()
   259→   259→    if (stored.size > 0) return stored
   260→   260→    // First visit: expand the active group (falls back to Home).
   261→   261→    const activeGroup = nav.find((g) => g.items.some((i) => i.key === current))?.group
   262→   262→    return new Set([activeGroup ?? 'Home'])
   263→   263→  })
   264→   264→
   265→   265→  // Keep the active route's group expanded (also covers search/quick-create jumps).
   266→   266→  // Derived-state adjustment during render — the React-recommended pattern.
   267→   267→  const [prevCurrent, setPrevCurrent] = useState(current)
   268→   268→  if (prevCurrent !== current) {
   269→   269→    setPrevCurrent(current)
   270→   270→    const group = nav.find((g) => g.items.some((i) => i.key === current))?.group
   271→   271→    if (group && !openGroups.has(group)) {
   272→   272→      const next = new Set(openGroups)
   273→   273→      next.add(group)
   274→   274→      persistOpenGroups(next)
   275→   275→      setOpenGroups(next)
   276→   276→    }
   277→   277→  }
   278→   278→
   279→   279→  function toggleGroup(group: string) {
   280→   280→    setOpenGroups((prev) => {
   281→   281→      const next = new Set(prev)
   282→   282→      if (next.has(group)) next.delete(group)
   283→   283→      else next.add(group)
   284→   284→      persistOpenGroups(next)
   285→   285→      return next
   286→   286→    })
   287→   287→  }
   288→   288→
   289→   289→  return (
   290→   290→    <div className="flex h-full flex-col">
   291→   291→      <div className="px-4 py-4">
   292→   292→        <Logo />
   293→   293→      </div>
   294→   294→      <ScrollArea className="flex-1 px-2 pb-4">
   295→   295→        <nav aria-label="ERP modules" className="space-y-0.5">
   296→   296→          {nav.map((g) => {
   297→   297→            const open = openGroups.has(g.group)
   298→   298→            const activeChild = g.items.some((i) => i.key === current)
   299→   299→            return (
   300→   300→              <div key={g.group}>
   301→   301→                <button
   302→   302→                  type="button"
   303→   303→                  onClick={() => toggleGroup(g.group)}
   304→   304→                  aria-expanded={open}
   305→   305→                  aria-controls={groupSlug(g.group)}
   306→   306→                  className={cn(
   307→   307→                    'mt-1 flex w-full items-center gap-2.5 rounded-md px-2.5 py-2 text-sm font-medium transition-colors hover:bg-accent focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring',
   308→   308→                    activeChild && !open
   309→   309→                      ? 'text-foreground'
   310→   310→                      : 'text-foreground/75',
   311→   311→                  )}
   312→   312→                >
   313→   313→                  <g.icon
   314→   314→                    className={cn(
   315→   315→                      'h-4 w-4 shrink-0',
   316→   316→                      activeChild ? 'text-primary' : 'text-muted-foreground',
   317→   317→                    )}
   318→   318→                    aria-hidden="true"
   319→   319→                  />
   320→   320→                  <span className="flex-1 truncate text-left">{g.group}</span>
   321→   321→                  <ChevronDown
   322→   322→                    className={cn(
   323→   323→                      'h-4 w-4 shrink-0 text-muted-foreground transition-transform duration-200',
   324→   324→                      open && 'rotate-180',
   325→   325→                    )}
   326→   326→                    aria-hidden="true"
   327→   327→                  />
   328→   328→                </button>
   329→   329→                {open && (
   330→   330→                  <ul id={groupSlug(g.group)} className="ml-[26px] mt-0.5 space-y-0.5 border-l pl-2">
   331→   331→                    {g.items.map((item) => {
   332→   332→                      const active = current === item.key
   333→   333→                      return (
   334→   334→                        <li key={item.key}>
   335→   335→                          <button
   336→   336→                            type="button"
   337→   337→                            onClick={() => navigate(item.key)}
   338→   338→                            aria-current={active ? 'page' : undefined}
   339→   339→                            className={cn(
   340→   340→                              'flex w-full items-center gap-2.5 rounded-md px-2.5 py-2 text-sm transition-colors',
   341→   341→                              active
   342→   342→                                ? 'bg-primary font-medium text-primary-foreground shadow-sm'
   343→   343→                                : 'text-foreground/80 hover:bg-accent hover:text-foreground',
   344→   344→                            )}
   345→   345→                          >
   346→   346→                            <item.icon className="h-4 w-4 shrink-0" aria-hidden="true" />
   347→   347→                            <span className="truncate">{item.label}</span>
   348→   348→                          </button>
   349→   349→                        </li>
   350→   350→                      )
   351→   351→                    })}
   352→   352→                  </ul>
   353→   353→                )}
   354→   354→              </div>
   355→   355→            )
   356→   356→          })}
   357→   357→        </nav>
   358→   358→      </ScrollArea>
   359→   359→      <div className="border-t px-4 py-3 text-[11px] text-muted-foreground">
   360→   360→        Rebuilt from dolibarr.org · v1.0
   361→   361→      </div>
   362→   362→    </div>
   363→   363→  )
   364→   364→}

   604→   604→/**
   605→   605→ * Default export — the auth-aware root.
   606→   606→ * loading → splash · anon → AuthScreen · authed → ErpApp (fresh per user).
   607→   607→ */
   608→   608→export default function ErpRoot() {
   609→   609→  const status = useAuthStore((s) => s.status)
   610→   610→  const user = useAuthStore((s) => s.user)
   611→   611→  const init = useAuthStore((s) => s.init)
   612→   612→  const forceSignOut = useAuthStore((s) => s.forceSignOut)
   613→   613→
   614→   614→  // Restore the session once on mount
   615→   615→  useEffect(() => {
   616→   616→    void init()
   617→   617→  }, [init])
   618→   618→
   619→   619→  // Session watchdog: immediate check + every 60s + on focus/visibility.
   620→   620→  // Silent while authed — only a 401 drops the local session.
   621→   621→  useEffect(() => {
   622→   622→    if (status !== 'authed') return
   623→   623→    let alive = true
   624→   624→    const check = async () => {
   625→   625→      try {
   626→   626→        const res = await fetch('/api/auth/me', {
   627→   627→          credentials: 'same-origin',
   628→   628→          cache: 'no-store',
   629→   629→          headers: authHeaders(),
   630→   630→        })
   631→   631→        if (alive && res.status === 401) {
   632→   632→          forceSignOut()
   633→   633→          window.location.hash = '#/dashboard'
   634→   634→        }
   635→   635→      } catch {
   636→   636→        // network hiccup — keep the current session state
   637→   637→      }
   638→   638→    }
   639→   639→    void check()
   640→   640→    const id = window.setInterval(() => { void check() }, 60_000)
   641→   641→    const onVisible = () => {
   642→   642→      if (document.visibilityState === 'visible') void check()
   643→   643→    }
   644→   644→    window.addEventListener('focus', onVisible)
   645→   645→    document.addEventListener('visibilitychange', onVisible)
   646→   646→    return () => {
   647→   647→      alive = false
   648→   648→      window.clearInterval(id)
   649→   649→      window.removeEventListener('focus', onVisible)
   650→   650→      document.removeEventListener('visibilitychange', onVisible)
   651→   651→    }
   652→   652→  }, [status, forceSignOut])
   653→   653→
   654→   654→  if (status === 'loading') return <Splash />
   655→   655→  if (status === 'anon' || !user) return <AuthScreen />
   656→   656→  return <ErpApp key={user.id} user={user} />
   657→   657→}

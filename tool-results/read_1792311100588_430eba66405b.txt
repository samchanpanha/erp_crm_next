   117→   117→const round2 = (n: number) => Math.round(n * 100) / 100
   118→   118→
   119→   119→function qs(params: Record<string, string>) {
   120→   120→  const sp = new URLSearchParams()
   121→   121→  for (const [k, v] of Object.entries(params)) if (v && v !== 'ALL') sp.set(k, v)
   122→   122→  const s = sp.toString()
   123→   123→  return s ? `?${s}` : ''
   124→   124→}
   125→   125→
   126→   126→function normalizeList<T>(raw: unknown, keys: string[]): T[] {
   127→   127→  if (Array.isArray(raw)) return raw as T[]
   128→   128→  if (raw && typeof raw === 'object') {
   129→   129→    const obj = raw as Record<string, unknown>
   130→   130→    for (const k of keys) if (Array.isArray(obj[k])) return obj[k] as T[]
   131→   131→  }
   132→   132→  return []
   133→   133→}

   135→   135→const emptyForm = (): DocForm => ({
   136→   136→  thirdPartyId: '',

   139→   139→  notes: '',
   140→   140→  lines: [{ label: '', qty: 1, unitPrice: 0, discountPct: 0, vatRate: 20 }],
   141→   141→})
   142→   142→
   143→   143→const toEditorLines = (lines: DocLine[] | undefined): EditorLine[] =>
   144→   144→  (lines ?? []).map((l) => ({
   145→   145→    productId: l.productId ?? undefined,
   146→   146→    label: l.label,
   147→   147→    qty: l.qty,
   148→   148→    unitPrice: l.unitPrice,
   149→   149→    discountPct: l.discountPct,
   150→   150→    vatRate: l.vatRate,
   151→   151→  }))
   152→   152→
   153→   153→const remainingOf = (i: { total: number; paidAmount: number }) => round2(i.total - i.paidAmount)
   154→   154→
   155→   155→/** True when an unpaid invoice is past its due date */
   156→   156→function isOverdue(i: { status: string; dueDate: string | null }): boolean {
   157→   157→  if (!i.dueDate || !['UNPAID', 'PARTIAL'].includes(i.status)) return false
   158→   158→  const today = new Date()
   159→   159→  today.setHours(0, 0, 0, 0)
   160→   160→  return new Date(i.dueDate) < today
   161→   161→}

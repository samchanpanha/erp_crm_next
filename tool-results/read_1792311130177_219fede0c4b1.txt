   142→function round2(n: number): number {
   143→  return Math.round(n * 100) / 100
   144→}
   145→
   146→function isOverdue(inv: SupplierInvoice, now = Date.now()): boolean {
   147→  return (
   148→    ['UNPAID', 'PARTIAL'].includes(inv.status) &&
   149→    !!inv.dueDate &&
   150→    new Date(inv.dueDate).getTime() < now
   151→  )
   152→}

   157→function SIFormInner({
   158→  editInvoice,
   159→  suppliers,
   160→  receivedOrders,
   161→  saving,
   162→  onSubmit,
   163→  onCancel,
   164→}: {
   165→  editInvoice: SupplierInvoice | null
   166→  suppliers: ThirdPartyLite[]
   167→  receivedOrders: PurchaseOrderLite[]
   168→  saving: boolean
   169→  onSubmit: (payload: unknown) => void
   170→  onCancel: () => void
   171→}) {
   172→  const [supplierId, setSupplierId] = useState(editInvoice?.thirdParty.id ?? '')
   173→  const [date, setDate] = useState(editInvoice ? toDateInput(editInvoice.date) : toDateInput(new Date()))
   174→  const [dueDate, setDueDate] = useState(editInvoice?.dueDate ? toDateInput(editInvoice.dueDate) : '')
   175→  const [subtotal, setSubtotal] = useState(editInvoice ? String(editInvoice.subtotal) : '0')
   176→  const [vatTotal, setVatTotal] = useState(editInvoice ? String(editInvoice.vatTotal) : '0')
   177→  const [purchaseOrderId, setPurchaseOrderId] = useState(editInvoice?.purchaseOrder?.id ?? NO_PO)
   178→  const [notes, setNotes] = useState(editInvoice?.notes ?? '')
   179→
   180→  const nSubtotal = Number(subtotal) || 0
   181→  const nVat = Number(vatTotal) || 0
   182→  const computedTotal = round2(nSubtotal + nVat)
   183→  const invalid = !supplierId || nSubtotal < 0 || nVat < 0
   184→
   185→  const submit = () => {
   186→    if (!supplierId) return
   187→    onSubmit({
   188→      thirdPartyId: supplierId,
   189→      date: date || undefined,
   190→      dueDate: dueDate || null,
   191→      subtotal: nSubtotal,
   192→      vatTotal: nVat,
   193→      notes: notes || null,
   194→      purchaseOrderId: purchaseOrderId === NO_PO ? null : purchaseOrderId,
   195→    })
   196→  }
   197→
   198→  return (
   199→    <>
   200→      <div className="space-y-4">
   201→        <div className="grid gap-3 sm:grid-cols-2">
   202→          <div className="space-y-1.5">
   203→            <Label htmlFor="si-supplier">Supplier *</Label>
   204→            <Select value={supplierId} onValueChange={setSupplierId}>
   205→              <SelectTrigger id="si-supplier" aria-label="Supplier" className="min-h-[40px]">
   206→                <SelectValue placeholder="Select supplier" />
   207→              </SelectTrigger>
   208→              <SelectContent>
   209→                {suppliers.map((s) => (
   210→                  <SelectItem key={s.id} value={s.id}>
   211→                    {s.name}
   212→                  </SelectItem>
   213→                ))}
   214→              </SelectContent>
   215→            </Select>
   216→          </div>
   217→          <div className="space-y-1.5">
   218→            <Label htmlFor="si-po">Linked purchase order</Label>
   219→            <Select value={purchaseOrderId} onValueChange={setPurchaseOrderId}>
   220→              <SelectTrigger id="si-po" aria-label="Linked purchase order" className="min-h-[40px]">
   221→                <SelectValue placeholder="None" />
   222→              </SelectTrigger>
   223→              <SelectContent>
   224→                <SelectItem value={NO_PO}>None</SelectItem>
   225→                {receivedOrders.map((o) => (
   226→                  <SelectItem key={o.id} value={o.id}>
   227→                    {o.ref} — {o.thirdParty.name} ({formatMoney(o.total)})
   228→                  </SelectItem>
   229→                ))}
   230→              </SelectContent>
   231→            </Select>
   232→          </div>
   233→          <div className="space-y-1.5">
   234→            <Label htmlFor="si-date">Invoice date</Label>
   235→            <Input id="si-date" type="date" value={date} onChange={(e) => setDate(e.target.value)} />
   236→          </div>
   237→          <div className="space-y-1.5">
   238→            <Label htmlFor="si-due">Payment due</Label>
   239→            <Input id="si-due" type="date" value={dueDate} onChange={(e) => setDueDate(e.target.value)} />
   240→          </div>
   241→          <div className="space-y-1.5">
   242→            <Label htmlFor="si-subtotal">Subtotal (net) *</Label>
   243→            <Input
   244→              id="si-subtotal"
   245→              type="number"
   246→              min="0"
   247→              step="0.01"
   248→              value={subtotal}
   249→              onChange={(e) => setSubtotal(e.target.value)}
   250→            />
   251→          </div>
   252→          <div className="space-y-1.5">
   253→            <Label htmlFor="si-vat">VAT total *</Label>
   254→            <Input
   255→              id="si-vat"
   256→              type="number"
   257→              min="0"
   258→              step="0.01"
   259→              value={vatTotal}
   260→              onChange={(e) => setVatTotal(e.target.value)}
   261→            />
   262→          </div>
   263→        </div>
   264→
   265→        <div className="flex items-center justify-between rounded-md border bg-muted/40 px-4 py-3">
   266→          <span className="text-sm text-muted-foreground">Total (subtotal + VAT)</span>
   267→          <span className="text-lg font-bold tabular-nums text-primary">{formatMoney(computedTotal)}</span>
   268→        </div>
   269→
   270→        <div className="space-y-1.5">
   271→          <Label htmlFor="si-notes">Notes</Label>
   272→          <Textarea id="si-notes" rows={2} placeholder="Internal notes (optional)" value={notes} onChange={(e) => setNotes(e.target.value)} />
   273→        </div>
   274→      </div>
   275→
   276→      <DialogFooter className="gap-2">
   277→        <Button variant="outline" onClick={onCancel}>Cancel</Button>
   278→        <Button onClick={submit} disabled={saving || invalid} className="min-h-[40px] min-w-[120px]">
   279→          {saving ? 'Saving…' : editInvoice ? 'Save changes' : 'Create draft'}
   280→        </Button>
   281→      </DialogFooter>
   282→    </>
   283→  )
   284→}

   286→function SIFormDialog({
   287→  open,
   288→  onOpenChange,
   289→  suppliers,
   290→  receivedOrders,
   291→  editId,
   292→  editInvoice,
   293→  toast,
   294→}: {
   295→  open: boolean
   296→  onOpenChange: (o: boolean) => void
   297→  suppliers: ThirdPartyLite[]
   298→  receivedOrders: PurchaseOrderLite[]
   299→  editId: string | null
   300→  editInvoice: SupplierInvoice | null
   301→  toast: ReturnType<typeof useToast>['toast']
   302→}) {
   303→  const queryClient = useQueryClient()
   304→
   305→  const saveMutation = useMutation({
   306→    mutationFn: (payload: unknown) =>
   307→      editId
   308→        ? put<SupplierInvoice>(`/api/supplier-invoices/${editId}`, payload)
   309→        : post<SupplierInvoice>('/api/supplier-invoices', payload),
   310→    onSuccess: (saved) => {
   311→      queryClient.invalidateQueries({ queryKey: ['supplier-invoices'] })
   312→      if (editId) queryClient.invalidateQueries({ queryKey: ['supplier-invoice', editId] })
   313→      toast({ title: editId ? 'Supplier invoice updated' : 'Supplier invoice created', description: saved.ref })
   314→      onOpenChange(false)
   315→    },
   316→    onError: (e: Error) =>
   317→      toast({ title: 'Save failed', description: e.message, variant: 'destructive' }),
   318→  })
   319→
   320→  const loadingEdit = !!editId && !editInvoice
   321→
   322→  return (
   323→    <Dialog open={open} onOpenChange={onOpenChange}>
   324→      <DialogContent className="max-h-[70vh] overflow-y-auto sm:max-w-2xl">
   325→        <DialogHeader>
   326→          <DialogTitle>{editId ? `Edit ${editInvoice?.ref ?? 'supplier invoice'}` : 'New supplier invoice'}</DialogTitle>
   327→          <DialogDescription>
   328→            {editId
   329→              ? 'Draft supplier invoices only — totals are recomputed on save.'
   330→              : 'Enter the invoice totals directly (no line items).'}
   331→          </DialogDescription>
   332→        </DialogHeader>
   333→
   334→        {loadingEdit ? (
   335→          <div className="space-y-3 py-6">
   336→            <Skeleton className="h-10 w-full" />
   337→            <Skeleton className="h-10 w-2/3" />
   338→            <Skeleton className="h-10 w-full" />
   339→          </div>
   340→        ) : (
   341→          <SIFormInner
   342→            key={editId ?? 'new'}
   343→            editInvoice={editInvoice}
   344→            suppliers={suppliers}
   345→            receivedOrders={receivedOrders}
   346→            saving={saveMutation.isPending}
   347→            onSubmit={(payload) => saveMutation.mutate(payload)}
   348→            onCancel={() => onOpenChange(false)}
   349→          />
   350→        )}
   351→      </DialogContent>
   352→    </Dialog>
   353→  )
   354→}

   356→// ---------------------------------------------------------------------------

   359→function PaymentDialog({
   360→  invoice,
   361→  bankAccounts,
   362→  open,
   363→  onOpenChange,
   364→  toast,
   365→}: {
   366→  invoice: SupplierInvoice
   367→  bankAccounts: BankAccountLite[]
   368→  open: boolean
   369→  onOpenChange: (o: boolean) => void
   370→  toast: ReturnType<typeof useToast>['toast']
   371→}) {
   372→  const queryClient = useQueryClient()
   373→  const remaining = round2(invoice.total - invoice.paidAmount)
   374→  const [amount, setAmount] = useState(String(remaining))
   375→  const [method, setMethod] = useState('BANK_TRANSFER')
   376→  const [date, setDate] = useState(toDateInput(new Date()))
   377→  const [reference, setReference] = useState('')
   378→  const [bankAccountId, setBankAccountId] = useState(NO_BANK)
   379→  const [note, setNote] = useState('')
   380→
   381→  const payMutation = useMutation({
   382→    mutationFn: (payload: unknown) => post<PaymentLite>('/api/payments', payload),
   383→    onSuccess: () => {
   384→      queryClient.invalidateQueries({ queryKey: ['supplier-invoices'] })
   385→      queryClient.invalidateQueries({ queryKey: ['supplier-invoice', invoice.id] })
   386→      toast({ title: 'Payment recorded', description: `${formatMoney(Number(amount))} applied to ${invoice.ref}` })
   387→      onOpenChange(false)
   388→    },
   389→    onError: (e: Error) =>
   390→      toast({ title: 'Payment failed', description: e.message, variant: 'destructive' }),
   391→  })
   392→
   393→  const submit = () => {
   394→    const nAmount = Number(amount)
   395→    if (!Number.isFinite(nAmount) || nAmount <= 0) {
   396→      toast({ title: 'Invalid amount', description: 'Payment amount must be greater than 0.', variant: 'destructive' })
   397→      return
   398→    }
   399→    payMutation.mutate({
   400→      supplierInvoiceId: invoice.id,
   401→      amount: nAmount,
   402→      method,
   403→      date: date || undefined,
   404→      reference: reference || undefined,
   405→      note: note || undefined,
   406→      bankAccountId: bankAccountId === NO_BANK ? undefined : bankAccountId,
   407→    })
   408→  }
   409→
   410→  return (
   411→    <Dialog open={open} onOpenChange={onOpenChange}>
   412→      <DialogContent className="max-h-[70vh] overflow-y-auto sm:max-w-lg">
   413→        <DialogHeader>
   414→          <DialogTitle>Record payment — {invoice.ref}</DialogTitle>
   415→          <DialogDescription>
   416→            {invoice.thirdParty.name} · Total {formatMoney(invoice.total)} · Remaining{' '}
   417→            <span className="font-semibold text-primary">{formatMoney(remaining)}</span>
   418→          </DialogDescription>
   419→        </DialogHeader>
   420→        <div className="grid gap-3 sm:grid-cols-2">
   421→          <div className="space-y-1.5">
   422→            <Label htmlFor="pay-amount">Amount *</Label>
   423→            <Input
   424→              id="pay-amount"
   425→              type="number"
   426→              min="0"
   427→              step="0.01"
   428→              value={amount}
   429→              onChange={(e) => setAmount(e.target.value)}
   430→            />
   431→          </div>
   432→          <div className="space-y-1.5">
   433→            <Label htmlFor="pay-method">Method *</Label>
   434→            <Select value={method} onValueChange={setMethod}>
   435→              <SelectTrigger id="pay-method" aria-label="Payment method" className="min-h-[40px]">
   436→                <SelectValue />
   437→              </SelectTrigger>
   438→              <SelectContent>
   439→                {PAYMENT_METHODS.map((m) => (
   440→                  <SelectItem key={m.value} value={m.value}>
   441→                    {m.label}
   442→                  </SelectItem>
   443→                ))}
   444→              </SelectContent>
   445→            </Select>
   446→          </div>
   447→          <div className="space-y-1.5">
   448→            <Label htmlFor="pay-date">Payment date</Label>
   449→            <Input id="pay-date" type="date" value={date} onChange={(e) => setDate(e.target.value)} />
   450→          </div>
   451→          <div className="space-y-1.5">
   452→            <Label htmlFor="pay-ref">Reference</Label>
   453→            <Input id="pay-ref" placeholder="e.g. TRX-99120" value={reference} onChange={(e) => setReference(e.target.value)} />
   454→          </div>
   455→          <div className="space-y-1.5 sm:col-span-2">
   456→            <Label htmlFor="pay-bank">Bank account (optional)</Label>
   457→            <Select value={bankAccountId} onValueChange={setBankAccountId}>
   458→              <SelectTrigger id="pay-bank" aria-label="Bank account" className="min-h-[40px]">
   459→                <SelectValue placeholder="None" />
   460→              </SelectTrigger>
   461→              <SelectContent>
   462→                <SelectItem value={NO_BANK}>None</SelectItem>
   463→                {bankAccounts.map((b) => (
   464→                  <SelectItem key={b.id} value={b.id}>
   465→                    {b.name}
   466→                  </SelectItem>
   467→                ))}
   468→              </SelectContent>
   469→            </Select>
   470→            {bankAccounts.length === 0 && (
   471→              <p className="text-xs text-muted-foreground">No bank accounts available.</p>
   472→            )}
   473→          </div>
   474→          <div className="space-y-1.5 sm:col-span-2">
   475→            <Label htmlFor="pay-note">Note</Label>
   476→            <Input id="pay-note" placeholder="Optional note" value={note} onChange={(e) => setNote(e.target.value)} />
   477→          </div>
   478→        </div>
   479→        <DialogFooter className="gap-2">
   480→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   481→          <Button onClick={submit} disabled={payMutation.isPending} className="min-h-[40px] min-w-[140px]">
   482→            {payMutation.isPending ? 'Recording…' : 'Record payment'}
   483→          </Button>
   484→        </DialogFooter>
   485→      </DialogContent>
   486→    </Dialog>
   487→  )
   488→}

   585→              <p className="mb-2 text-sm font-semibold">Payments</p>

   642→            </div>
   643→          </div>

   645→      </DialogContent>

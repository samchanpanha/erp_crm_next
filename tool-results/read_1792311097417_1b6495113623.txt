    36→// ---------------------------------------------------------------------------
    37→// Types (mirror API payloads)
    38→// ---------------------------------------------------------------------------
    39→type ThirdPartyType = 'CUSTOMER' | 'PROSPECT' | 'SUPPLIER' | 'CUSTOMER_SUPPLIER'
    40→type ThirdPartyStatus = 'ACTIVE' | 'SUSPENDED'
    41→type PaymentTerms = 'ON_ORDER' | 'NET15' | 'NET30' | 'NET60'
    42→
    43→interface ThirdPartyCounts {
    44→  contacts: number
    45→  opportunities: number
    46→  proposals: number
    47→  saleOrders: number
    48→  invoices: number
    49→}
    50→
    51→interface ThirdPartyRow {
    52→  id: string
    53→  code: string
    54→  name: string
    55→  type: ThirdPartyType
    56→  status: ThirdPartyStatus
    57→  email: string | null
    58→  phone: string | null
    59→  website: string | null
    60→  address: string | null
    61→  zip: string | null
    62→  city: string | null
    63→  country: string | null
    64→  vatNumber: string | null
    65→  paymentTerms: PaymentTerms
    66→  notes: string | null
    67→  createdAt: string
    68→  _count: ThirdPartyCounts
    69→}
    70→
    71→interface ContactRow {
    72→  id: string
    73→  thirdPartyId: string | null
    74→  firstName: string
    75→  lastName: string
    76→  position: string | null
    77→  email: string | null
    78→  phone: string | null
    79→  mobile: string | null
    80→  isPrimary: boolean
    81→  createdAt: string
    82→}
    83→
    84→interface DetailCounts extends ThirdPartyCounts {
    85→  contracts: number
    86→  purchaseOrders: number
    87→  supplierInvoices: number
    88→  tickets: number
    89→  projects: number
    90→  agendaEvents: number
    91→}
    92→
    93→interface ThirdPartyDetail extends Omit<ThirdPartyRow, '_count'> {
    94→  contacts: ContactRow[]
    95→  _count: DetailCounts
    96→}

   148→// ---------------------------------------------------------------------------
   149→// Module
   150→// ---------------------------------------------------------------------------
   151→export default function ThirdPartiesModule() {
   152→  const { toast } = useToast()
   153→  const queryClient = useQueryClient()
   154→
   155→  // filters
   156→  const [search, setSearch] = useState('')
   157→  const [typeFilter, setTypeFilter] = useState<string>('ALL')
   158→  const [statusFilter, setStatusFilter] = useState<string>('ALL')
   159→
   160→  // dialogs
   161→  const [formOpen, setFormOpen] = useState(false)
   162→  const [editing, setEditing] = useState<ThirdPartyRow | null>(null)
   163→  const [form, setForm] = useState<TpFormState>(emptyTpForm)
   164→  const [detailId, setDetailId] = useState<string | null>(null)
   165→  const [confirm, setConfirm] = useState<ConfirmState>(null)
   166→  const [contactForm, setContactForm] = useState<ContactFormState>(emptyContactForm)
   167→
   168→  const listQuery = useQuery({
   169→    queryKey: ['third-parties'],
   170→    queryFn: () => get<ThirdPartyRow[]>('/api/third-parties'),
   171→  })
   172→
   173→  const detailQuery = useQuery({
   174→    queryKey: ['third-party', detailId],
   175→    queryFn: () => get<ThirdPartyDetail>(`/api/third-parties/${detailId}`),
   176→    enabled: detailId !== null,
   177→  })
   178→
   179→  const data = listQuery.data ?? []
   180→  const totalCounts = useMemo(
   181→    () => ({
   182→      CUSTOMER: data.filter((t) => t.type === 'CUSTOMER').length,
   183→      PROSPECT: data.filter((t) => t.type === 'PROSPECT').length,
   184→      SUPPLIER: data.filter((t) => t.type === 'SUPPLIER').length,
   185→    }),
   186→    [data],
   187→  )
   188→
   189→  const filtered = useMemo(() => {
   190→    const q = search.trim().toLowerCase()
   191→    return data.filter((tp) => {
   192→      if (typeFilter !== 'ALL' && tp.type !== typeFilter) return false
   193→      if (statusFilter !== 'ALL' && tp.status !== statusFilter) return false
   194→      if (!q) return true
   195→      return [tp.name, tp.code, tp.email ?? '', tp.city ?? ''].some((v) =>
   196→        v.toLowerCase().includes(q),
   197→      )
   198→    })
   199→  }, [data, search, typeFilter, statusFilter])
   200→
   201→  const invalidateAll = () => {
   202→    void queryClient.invalidateQueries({ queryKey: ['third-parties'] })
   203→    void queryClient.invalidateQueries({ queryKey: ['third-party'] })
   204→  }
   205→
   206→  // ---- mutations ----------------------------------------------------------
   207→  const saveMutation = useMutation({
   208→    mutationFn: async () => {
   209→      const payload = { ...form, name: form.name.trim() }
   210→      if (editing) return put(`/api/third-parties/${editing.id}`, payload)
   211→      return post('/api/third-parties', payload)
   212→    },
   213→    onSuccess: () => {
   214→      toast({
   215→        title: editing ? 'Third party updated' : 'Third party created',
   216→        description: form.name.trim(),
   217→      })
   218→      invalidateAll()
   219→      setFormOpen(false)
   220→      setEditing(null)
   221→    },
   222→    onError: (err: Error) =>
   223→      toast({ title: 'Save failed', description: err.message, variant: 'destructive' }),
   224→  })
   225→
   226→  const deleteTpMutation = useMutation({
   227→    mutationFn: (id: string) => del(`/api/third-parties/${id}`),
   228→    onSuccess: (_res, id) => {
   229→      toast({ title: 'Third party deleted' })
   230→      if (detailId === id) setDetailId(null)
   231→      invalidateAll()
   232→    },
   233→    onError: (err: Error) =>
   234→      toast({ title: 'Cannot delete', description: err.message, variant: 'destructive' }),
   235→  })
   236→
   237→  const addContactMutation = useMutation({
   238→    mutationFn: () =>
   239→      post('/api/contacts', { thirdPartyId: detailId, ...contactForm }),
   240→    onSuccess: () => {
   241→      toast({ title: 'Contact added', description: `${contactForm.firstName} ${contactForm.lastName}`.trim() })
   242→      setContactForm(emptyContactForm)
   243→      invalidateAll()
   244→    },
   245→    onError: (err: Error) =>
   246→      toast({ title: 'Could not add contact', description: err.message, variant: 'destructive' }),
   247→  })
   248→
   249→  const deleteContactMutation = useMutation({
   250→    mutationFn: (id: string) => del(`/api/contacts/${id}`),
   251→    onSuccess: () => {
   252→      toast({ title: 'Contact deleted' })
   253→      invalidateAll()
   254→    },
   255→    onError: (err: Error) =>
   256→      toast({ title: 'Could not delete contact', description: err.message, variant: 'destructive' }),
   257→  })
   258→
   259→  // ---- dialog helpers -----------------------------------------------------
   260→  const openCreate = () => {
   261→    setEditing(null)
   262→    setForm(emptyTpForm)
   263→    setFormOpen(true)
   264→  }
   265→

   267→    setEditing(tp)
   268→    setForm({
   269→      name: tp.name,
   270→      type: tp.type,
   271→      status: tp.status,
   272→      email: tp.email ?? '',
   273→      phone: tp.phone ?? '',
   274→      website: tp.website ?? '',
   275→      address: tp.address ?? '',
   276→      zip: tp.zip ?? '',
   277→      city: tp.city ?? '',
   278→      country: tp.country ?? '',
   279→      vatNumber: tp.vatNumber ?? '',
   280→      paymentTerms: tp.paymentTerms,
   281→      notes: tp.notes ?? '',
   282→    })
   283→    setFormOpen(true)
   284→  }
   285→
   286→  const detail = detailQuery.data
   287→  const canSaveTp = form.name.trim().length > 0 && !saveMutation.isPending
   288→  const canAddContact =
   289→    contactForm.firstName.trim().length > 0 &&
   290→    contactForm.lastName.trim().length > 0 &&
   291→    !addContactMutation.isPending
   292→
   293→  return (
   294→    <div className="space-y-5">
   295→      <PageHeader
   296→        icon={Building2}
   297→        title="Third Parties"
   298→        description="Customers, prospects and suppliers — the heart of the CRM"
   299→        actions={
   300→          <Button onClick={openCreate} className="min-h-11 sm:min-h-9">
   301→            <Plus className="h-4 w-4" aria-hidden />
   302→            New third party
   303→          </Button>
   304→        }
   305→      />
   306→
   307→      {/* Summary chips */}
   308→      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   309→        <StatCard title="Customers" value={totalCounts.CUSTOMER} icon={Building2} />
   310→        <StatCard title="Prospects" value={totalCounts.PROSPECT} icon={Target} />
   311→        <StatCard title="Suppliers" value={totalCounts.SUPPLIER} icon={Truck} />
   312→        <StatCard title="All third parties" value={data.length} icon={Users} hint="incl. customer + supplier" />
   313→      </div>
   314→
   315→      {/* Toolbar */}
   316→      <div className="flex flex-col gap-2 sm:flex-row sm:flex-wrap sm:items-center">
   317→        <div className="relative w-full sm:w-72">
   318→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" aria-hidden />
   319→          <Input
   320→            value={search}
   321→            onChange={(e) => setSearch(e.target.value)}
   322→            placeholder="Search name, code, email, city…"
   323→            className="pl-8"
   324→            aria-label="Search third parties"
   325→          />
   326→        </div>
   327→        <Select value={typeFilter} onValueChange={setTypeFilter}>
   328→          <SelectTrigger className="w-full sm:w-44" aria-label="Filter by type">
   329→            <SelectValue placeholder="Type" />
   330→          </SelectTrigger>
   331→          <SelectContent>
   332→            <SelectItem value="ALL">All types</SelectItem>
   333→            <SelectItem value="CUSTOMER">Customer</SelectItem>
   334→            <SelectItem value="PROSPECT">Prospect</SelectItem>
   335→            <SelectItem value="SUPPLIER">Supplier</SelectItem>
   336→            <SelectItem value="CUSTOMER_SUPPLIER">Customer + Supplier</SelectItem>
   337→          </SelectContent>
   338→        </Select>
   339→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   340→          <SelectTrigger className="w-full sm:w-40" aria-label="Filter by status">
   341→            <SelectValue placeholder="Status" />
   342→          </SelectTrigger>
   343→          <SelectContent>
   344→            <SelectItem value="ALL">All statuses</SelectItem>
   345→            <SelectItem value="ACTIVE">Active</SelectItem>
   346→            <SelectItem value="SUSPENDED">Suspended</SelectItem>
   347→          </SelectContent>
   348→        </Select>
   349→      </div>
   350→
   351→      {/* Table */}
   352→      <Card className="py-0 gap-0">
   353→        <CardContent className="p-0">
   354→          {listQuery.isError && (
   355→            <p className="p-4 text-sm text-destructive">
   356→              Failed to load third parties: {listQuery.error.message}
   357→            </p>
   358→          )}
   359→          <div className="overflow-x-auto">
   360→            <Table className="min-w-[900px]">
   361→              <TableHeader>
   362→                <TableRow>
   363→                  <TableHead>Code</TableHead>
   364→                  <TableHead>Name</TableHead>
   365→                  <TableHead>Type</TableHead>
   366→                  <TableHead>City / Country</TableHead>
   367→                  <TableHead>Phone</TableHead>
   368→                  <TableHead>Payment terms</TableHead>
   369→                  <TableHead className="text-center">Docs</TableHead>
   370→                  <TableHead>Status</TableHead>
   371→                  <TableHead className="text-right">Actions</TableHead>
   372→                </TableRow>
   373→              </TableHeader>
   374→              <TableBody>
   375→                {listQuery.isPending &&
   376→                  Array.from({ length: 5 }).map((_, i) => (
   377→                    <TableRow key={i}>
   378→                      <TableCell colSpan={9}>
   379→                        <Skeleton className="h-8 w-full" />
   380→                      </TableCell>
   381→                    </TableRow>
   382→                  ))}
   383→                {!listQuery.isPending && filtered.length === 0 && (
   384→                  <TableRow>
   385→                    <TableCell colSpan={9} className="p-0">
   386→                      <div className="p-6">
   387→                        <EmptyState
   388→                          icon={Building2}
   389→                          title={data.length === 0 ? 'No third parties yet' : 'No third parties match your filters'}
   390→                          description={
   391→                            data.length === 0
   392→                              ? 'Create your first customer, prospect or supplier to get started.'
   393→                              : 'Try clearing the search box or switching the type/status filters.'
   394→                          }
   395→                          actionLabel={data.length === 0 ? 'New third party' : undefined}
   396→                          onAction={data.length === 0 ? openCreate : undefined}
   397→                        />
   398→                      </div>
   399→                    </TableCell>
   400→                  </TableRow>
   401→                )}
   402→                {filtered.map((tp) => (
   403→                  <TableRow key={tp.id} className="group">
   404→                    <TableCell className="whitespace-nowrap font-mono text-xs text-muted-foreground">
   405→                      {tp.code}
   406→                    </TableCell>
   407→                    <TableCell className="max-w-[220px]">
   408→                      <button
   409→                        type="button"
   410→                        onClick={() => setDetailId(tp.id)}
   411→                        className="rounded text-left font-medium hover:text-primary hover:underline focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   412→                        aria-label={`View details of ${tp.name}`}
   413→                      >
   414→                        {tp.name}
   415→                      </button>
   416→                      {tp.email && (
   417→                        <p className="truncate text-xs text-muted-foreground">{tp.email}</p>
   418→                      )}
   419→                    </TableCell>
   420→                    <TableCell>
   421→                      <StatusBadge status={tp.type} label={TYPE_LABEL[tp.type]} tone={TYPE_TONE[tp.type]} />
   422→                    </TableCell>
   423→                    <TableCell className="whitespace-nowrap text-sm">
   424→                      {[tp.city, tp.country].filter(Boolean).join(', ') || '—'}
   425→                    </TableCell>
   426→                    <TableCell className="whitespace-nowrap text-sm">{tp.phone ?? '—'}</TableCell>
   427→                    <TableCell className="whitespace-nowrap text-sm">
   428→                      {PAYMENT_TERMS_LABEL[tp.paymentTerms] ?? tp.paymentTerms}
   429→                    </TableCell>
   430→                    <TableCell className="text-center">
   431→                      <span
   432→                        className="inline-flex items-center gap-1 text-sm tabular-nums"
   433→                        title="Proposals, sale orders and invoices"
   434→                      >
   435→                        <FileText className="h-3.5 w-3.5 text-muted-foreground" aria-hidden />
   436→                        {tp._count.proposals + tp._count.saleOrders + tp._count.invoices}
   437→                      </span>
   438→                    </TableCell>
   439→                    <TableCell>
   440→                      <DocStatusBadge status={tp.status} />
   441→                    </TableCell>
   442→                    <TableCell>
   443→                      <div className="flex items-center justify-end gap-1">
   444→                        <Button
   445→                          variant="ghost"
   446→                          size="icon"
   447→                          className="h-9 w-9 sm:h-8 sm:w-8"
   448→                          onClick={() => setDetailId(tp.id)}
   449→                          aria-label={`View ${tp.name}`}
   450→                        >
   451→                          <Eye className="h-4 w-4" aria-hidden />
   452→                        </Button>
   453→                        <Button
   454→                          variant="ghost"
   455→                          size="icon"
   456→                          className="h-9 w-9 sm:h-8 sm:w-8"
   457→                          onClick={() => openEdit(tp)}
   458→                          aria-label={`Edit ${tp.name}`}
   459→                        >
   460→                          <Pencil className="h-4 w-4" aria-hidden />
   461→                        </Button>
   462→                        <Button
   463→                          variant="ghost"
   464→                          size="icon"
   465→                          className="h-9 w-9 text-muted-foreground hover:text-destructive sm:h-8 sm:w-8"
   466→                          onClick={() => setConfirm({ kind: 'tp', tp })}
   467→                          aria-label={`Delete ${tp.name}`}
   468→                        >
   469→                          <Trash2 className="h-4 w-4" aria-hidden />
   470→                        </Button>
   471→                      </div>
   472→                    </TableCell>
   473→                  </TableRow>
   474→                ))}
   475→              </TableBody>
   476→            </Table>
   477→          </div>
   478→          {!listQuery.isPending && filtered.length > 0 && (
   479→            <p className="border-t px-4 py-2 text-xs text-muted-foreground">
   480→              {filtered.length} of {data.length} third part{data.length === 1 ? 'y' : 'ies'}
   481→            </p>
   482→          )}
   483→        </CardContent>
   484→      </Card>
   485→
   486→      {/* Create / edit dialog */}
   487→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   488→        <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-xl">
   489→          <DialogHeader>
   490→            <DialogTitle>{editing ? `Edit ${editing.name}` : 'New third party'}</DialogTitle>
   491→            <DialogDescription>
   492→              {editing
   493→                ? `Reference ${editing.code} — update the company details below.`
   494→                : 'Register a customer, prospect or supplier. A reference code is generated automatically.'}
   495→            </DialogDescription>
   496→          </DialogHeader>
   497→
   498→          <div className="grid gap-4 sm:grid-cols-2">
   499→            <div className="space-y-1.5 sm:col-span-2">
   500→              <Label htmlFor="tp-name">
   501→                Name <span className="text-destructive">*</span>
   502→              </Label>
   503→              <Input
   504→                id="tp-name"
   505→                value={form.name}
   506→                onChange={(e) => setForm({ ...form, name: e.target.value })}
   507→                placeholder="Acme Manufacturing"
   508→                autoFocus
   509→              />
   510→            </div>
   511→            <div className="space-y-1.5">
   512→              <Label htmlFor="tp-type">Type</Label>
   513→              <Select
   514→                value={form.type}
   515→                onValueChange={(v) => setForm({ ...form, type: v as ThirdPartyType })}
   516→              >
   517→                <SelectTrigger id="tp-type" className="w-full">
   518→                  <SelectValue />
   519→                </SelectTrigger>
   520→                <SelectContent>
   521→                  <SelectItem value="CUSTOMER">Customer</SelectItem>
   522→                  <SelectItem value="PROSPECT">Prospect</SelectItem>
   523→                  <SelectItem value="SUPPLIER">Supplier</SelectItem>
   524→                  <SelectItem value="CUSTOMER_SUPPLIER">Customer + Supplier</SelectItem>
   525→                </SelectContent>
   526→              </Select>
   527→            </div>
   528→            <div className="space-y-1.5">
   529→              <Label htmlFor="tp-status">Status</Label>
   530→              <Select
   531→                value={form.status}
   532→                onValueChange={(v) => setForm({ ...form, status: v as ThirdPartyStatus })}
   533→              >
   534→                <SelectTrigger id="tp-status" className="w-full">
   535→                  <SelectValue />

   550→                placeholder="contact@company.com"
   551→              />
   552→            </div>
   553→            <div className="space-y-1.5">
   554→              <Label htmlFor="tp-phone">Phone</Label>
   555→              <Input
   556→                id="tp-phone"

   868→          </AlertDialogFooter>

   871→    </div>

   878→function InfoLine({
   879→  icon: Icon,
   880→  label,
   881→  value,
   882→  href,
   883→}: {
   884→  icon: typeof Mail
   885→  label: string
   886→  value: string | null
   887→  href?: string
   888→}) {
   889→  return (
   890→    <div className="flex items-start gap-2 min-w-0">
   891→      <Icon className="mt-0.5 h-4 w-4 shrink-0 text-muted-foreground" aria-hidden />
   892→      <div className="min-w-0">
   893→        <p className="text-xs uppercase tracking-wide text-muted-foreground">{label}</p>
   894→        {value ? (
   895→          href ? (
   896→            <a
   897→              href={href}
   898→              target="_blank"
   899→              rel="noreferrer"
   900→              className="break-all text-sm text-primary hover:underline"
   901→            >
   902→              {value}
   903→            </a>
   904→          ) : (
   905→            <p className="break-all text-sm">{value}</p>
   906→          )
   907→        ) : (
   908→          <p className="text-sm text-muted-foreground">—</p>
   909→        )}
   910→      </div>
   911→    </div>
   912→  )
   913→}

   915→function CountBox({ label, value }: { label: string; value: number }) {
   916→  return (
   917→    <div
   918→      className={cn(
   919→        'rounded-md border px-2 py-2 text-center',
   920→        value === 0 && 'opacity-50',
   921→      )}
   922→    >
   923→      <p className="text-base font-bold tabular-nums">{value}</p>
   924→      <p className="truncate text-[11px] text-muted-foreground">{label}</p>
   925→    </div>
   926→  )
   927→}

    92→    92→/** Margin % for products with a real cost; null renders as "—" */
    93→    93→function marginPct(p: ProductRow): number | null {
    94→    94→  if (p.type !== 'PRODUCT' || p.costPrice <= 0 || p.sellPrice <= 0) return null
    95→    95→  return ((p.sellPrice - p.costPrice) / p.sellPrice) * 100
    96→    96→}

   124→   124→      (p) => p.type === 'PRODUCT' && p.manageStock && p.stockQty <= p.minStock,
   125→   125→    ).length

   143→   143→    mutationFn: async () => {
   144→   144→      const payload: Record<string, unknown> = {
   145→   145→        label: form.label.trim(),
   146→   146→        type: form.type,

   149→   149→        costPrice: Number(form.costPrice) || 0,

   194→   194→  })

   200→   200→    setDialogOpen(true)

   206→   206→      label: p.label,
   207→   207→      type: p.type as ProductType,
   208→   208→      description: p.description ?? '',
   209→   209→      sellPrice: String(p.sellPrice),
   210→   210→      costPrice: String(p.costPrice),
   211→   211→      vatRate: String(p.vatRate),
   212→   212→      unit: p.unit,
   213→   213→      barcode: p.barcode ?? '',
   214→   214→      manageStock: p.manageStock,
   215→   215→      stockQty: String(p.stockQty),
   216→   216→      minStock: String(p.minStock),
   217→   217→      status: p.status as ProductStatus,
   218→   218→    })

   220→   220→    setDialogOpen(true)

   232→   232→    setFormError(null)
   233→   233→    saveMutation.mutate()

   235→   235→
   236→   236→  const set = <K extends keyof ProductForm>(key: K, value: ProductForm[K]) =>

   240→   240→  const trackStock = !isService && form.manageStock

   242→   242→  return (

   257→   257→
   258→   258→      {/* Stats */}

   265→   265→            <StatCard title="Services" value={stats.services} hint="Non-stock catalog items" icon={Wrench} />
   266→   266→            <StatCard title="Stock value (cost)" value={stats.stockValue} isMoney hint="Σ stock qty × cost price" icon={Wallet} />

   285→   285→            <TabsTrigger value="low" className="gap-1.5">

   292→   292→            </TabsTrigger>
   293→   293→          </TabsList>
   294→   294→        </Tabs>
   295→   295→        <div className="relative w-full lg:max-w-xs">

   297→   297→          <Input

   299→   299→            onChange={(e) => setSearch(e.target.value)}

   321→   321→                description={
   322→   322→                  search || tab !== 'all'
   323→   323→                    ? 'Try a different search term or tab.'

   327→   327→                onAction={openCreate}

   333→   333→                <TableHeader>
   334→   334→                  <TableRow>

   338→   338→                    <TableHead className="text-right">Sell price</TableHead>

   417→   417→                        </TableCell>
   418→   418→                      </TableRow>
   419→   419→                    )
   420→   420→                  })}
   421→   421→                </TableBody>
   422→   422→              </Table>
   423→   423→            </div>
   424→   424→          )}
   425→   425→        </CardContent>
   426→   426→      </Card>
   427→   427→
   428→   428→      {/* Create / Edit dialog */}
   429→   429→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   430→   430→        <DialogContent className="sm:max-w-xl">
   431→   431→          <DialogHeader>
   432→   432→            <DialogTitle>{editing ? 'Edit product' : 'New product or service'}</DialogTitle>
   433→   433→            <DialogDescription>
   434→   434→              {editing

   437→   437→            </DialogDescription>
   438→   438→          </DialogHeader>
   439→   439→
   440→   440→          <div className="max-h-[70vh] overflow-y-auto pr-1">

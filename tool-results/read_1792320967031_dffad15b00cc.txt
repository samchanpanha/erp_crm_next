   108→function parseConfig(raw: string | undefined): TemplateConfig {
   109→  if (!raw) return DEFAULT_TEMPLATE_CONFIG
   110→  try {
   111→    const parsed = templateConfigSchema.safeParse(JSON.parse(raw))
   112→    return parsed.success ? parsed.data : DEFAULT_TEMPLATE_CONFIG
   113→  } catch {
   114→    return DEFAULT_TEMPLATE_CONFIG
   115→  }
   116→}
   117→
   118→function useDocTemplate(kind: DocKind, enabled: boolean) {
   119→  return useQuery({
   120→    queryKey: ['doc-template', kind],
   121→    enabled,
   122→    staleTime: 60_000,
   123→    queryFn: async (): Promise<{ name: string; config: TemplateConfig }> => {
   124→      try {
   125→        const rows = await get<TemplateRow[]>(`/api/doc-templates?type=${kind}`)
   126→        const row = rows.find((r) => r.isDefault) ?? rows[0]
   127→        if (!row) return { name: 'Default template', config: DEFAULT_TEMPLATE_CONFIG }
   128→        return { name: row.name, config: parseConfig(row.config) }
   129→      } catch {
   130→        return { name: 'Default template', config: DEFAULT_TEMPLATE_CONFIG }
   131→      }
   132→    },
   133→  })
   134→}

   136→// ---------------------------------------------------------------------------
   137→// Print plumbing — body.printing-doc + @page size style + window.print()
   138→// ---------------------------------------------------------------------------
   139→export function usePrintNow(paperSize: 'A4' | 'LETTER' = 'A4') {
   140→  return useCallback(() => {
   141→    document.body.classList.add('printing-doc')
   142→
   143→    // Temporary @page + dialog-flattening rules (removed again on cleanup).
   144→    document.getElementById('print-page-size')?.remove()
   145→    const style = document.createElement('style')
   146→    style.id = 'print-page-size'
   147→    style.textContent = [
   148→      `@page { size: ${paperSize === 'LETTER' ? 'letter' : 'A4'}; margin: 12mm }`,
   149→      // The preview is rendered inside a fixed + transformed Radix dialog —
   150→      // flatten it so .print-area's absolute positioning resolves to the page.
   151→      'body.printing-doc [data-slot="dialog-content"] {',
   152→      '  position: static !important; transform: none !important; inset: auto !important;',
   153→      '  width: auto !important; max-width: none !important; margin: 0 !important; padding: 0 !important;',
   154→      '  border: 0 !important; box-shadow: none !important; background: white !important;',
   155→      '  overflow: visible !important; display: block !important;',
   156→      '}',
   157→      'body.printing-doc [data-slot="dialog-overlay"] { display: none !important; }',
   158→      'body.printing-doc .print-area { max-height: none !important; width: auto !important; max-width: 100% !important; box-sizing: border-box !important; }',
   159→    ].join('\n')
   160→    document.head.appendChild(style)
   161→
   162→    const cleanup = () => {
   163→      document.body.classList.remove('printing-doc')
   164→      document.getElementById('print-page-size')?.remove()
   165→    }
   166→    window.addEventListener('afterprint', cleanup, { once: true })
   167→    window.print()
   168→    // Safety net in case afterprint never fires (e.g. print dialog dismissed oddly)
   169→    window.setTimeout(cleanup, 2000)
   170→  }, [paperSize])
   171→}

   176→function companyAddressLine(s: SettingsMap): string {
   177→  const cityLine = [s['company.zip'], s['company.city']].filter(Boolean).join(' ')
   178→  return [s['company.address'], cityLine, s['company.country']]
   179→    .filter((p): p is string => !!p)
   180→    .join(', ')
   181→}
   182→
   183→function customerAddressLine(tp: {
   184→  address?: string | null
   185→  zip?: string | null
   186→  city?: string | null
   187→  country?: string | null
   188→}): string {
   189→  const cityLine = [tp.zip, tp.city].filter(Boolean).join(' ')
   190→  return [tp.address, cityLine, tp.country]
   191→    .filter((p): p is string => !!p && p.length > 0)
   192→    .join(', ')
   193→}

   224→function MetaBox({ rows }: { rows: { label: string; value: string }[] }) {
   225→  const visible = rows.filter((r) => r.label && r.value && r.value !== '—')
   226→  if (visible.length === 0) return null
   227→  return (
   228→    <div className="min-w-52 rounded-md border border-neutral-300 px-4 py-3 text-[12px]">
   229→      {visible.map((r) => (
   230→        <div key={r.label} className="flex justify-between gap-6 py-0.5">
   231→          <span className="text-neutral-500">{r.label}</span>
   232→          <span className="font-medium">{r.value}</span>
   233→        </div>
   234→      ))}
   235→    </div>
   236→  )
   237→}

   321→function buildInvoiceValues(d: InvoicePrintData, s: SettingsMap, docType: string): Record<string, string> {
   322→  const paid = d.paidAmount ?? 0
   323→  const due = Math.max(0, Math.round((d.total - paid) * 100) / 100)
   324→  return {
   325→    'company.name': s['company.name'] ?? '',
   326→    'company.legalName': s['company.legalName'] ?? '',
   327→    'company.address': companyAddressLine(s),
   328→    'company.phone': s['company.phone'] ?? '',
   329→    'company.email': s['company.email'] ?? '',
   330→    'company.website': s['company.website'] ?? '',
   331→    'company.taxId': s['company.taxId'] ?? '',
   332→    'company.bankDetails': s['company.bankDetails'] ?? '',
   333→    'doc.type': docType,
   334→    'doc.ref': d.ref,
   335→    'doc.date': formatDate(d.date),
   336→    'doc.dueDate': d.dueDate ? formatDate(d.dueDate) : '',
   337→    'doc.status': STATUS_LABELS[d.status] ?? d.status,
   338→    'doc.subtotal': formatMoney(d.subtotal),
   339→    'doc.discountTotal': formatMoney(d.discountTotal),
   340→    'doc.vatTotal': formatMoney(d.vatTotal),
   341→    'doc.total': formatMoney(d.total),
   342→    'doc.paidAmount': formatMoney(paid),
   343→    'doc.dueAmount': formatMoney(due),
   344→    'doc.notes': d.notes ?? '',
   345→    'customer.name': d.thirdParty.name,
   346→    'customer.code': d.thirdParty.code,
   347→    'customer.address': customerAddressLine(d.thirdParty),
   348→    'customer.email': d.thirdParty.email ?? '',
   349→    'customer.phone': d.thirdParty.phone ?? '',
   350→    'customer.taxId': d.thirdParty.taxId ?? '',
   351→  }
   352→}

   354→function buildReceiptValues(d: ReceiptPrintData, s: SettingsMap): Record<string, string> {
   355→  const remaining = Math.max(0, Math.round((d.invoice.total - d.invoice.paidAmount) * 100) / 100)
   356→  return {
   357→    'company.name': s['company.name'] ?? '',
   358→    'company.legalName': s['company.legalName'] ?? '',
   359→    'company.address': companyAddressLine(s),
   360→    'company.phone': s['company.phone'] ?? '',
   361→    'company.email': s['company.email'] ?? '',
   362→    'company.website': s['company.website'] ?? '',
   363→    'company.taxId': s['company.taxId'] ?? '',
   364→    'company.bankDetails': s['company.bankDetails'] ?? '',
   365→    'doc.type': DOC_TYPE_LABELS['RECEIPT'] ?? 'Payment receipt',
   366→    'doc.ref': d.invoice.ref,
   367→    'doc.date': formatDate(d.date),
   368→    'doc.dueDate': '',
   369→    'doc.status':
   370→      remaining <= 0.001 ? (STATUS_LABELS.PAID ?? 'Paid') : (STATUS_LABELS.PARTIAL ?? 'Partially paid'),
   371→    'doc.total': formatMoney(d.invoice.total),
   372→    'doc.paidAmount': formatMoney(d.invoice.paidAmount),
   373→    'doc.dueAmount': formatMoney(remaining),
   374→    'customer.name': d.customer.name,
   375→    'customer.code': d.customer.code,
   376→    'customer.address': customerAddressLine(d.customer),
   377→    'customer.email': d.customer.email ?? '',
   378→    'customer.phone': d.customer.phone ?? '',
   379→    'customer.taxId': d.customer.taxId ?? '',
   380→    'payment.amount': formatMoney(d.amount),
   381→    'payment.method': PAYMENT_METHOD_LABELS[d.method] ?? d.method,
   382→    'payment.date': formatDate(d.date),
   383→    'payment.reference': d.reference ?? d.paymentRef ?? '',
   384→    'payment.invoiceRef': d.invoice.ref,
   385→    'payment.remaining': formatMoney(remaining),
   386→  }
   387→}

   392→function LineDocView({
   393→  data,
   394→  config,
   395→  settings,
   396→  docType,
   397→  dueLabel,
   398→  billToLabel = 'Bill to',
   399→}: {
   400→  data: InvoicePrintData
   401→  config: TemplateConfig
   402→  settings: SettingsMap
   403→  docType: string
   404→  dueLabel: string
   405→  billToLabel?: string
   406→}) {
   407→  const accent = config.accentColor
   408→  const f = config.fields
   409→  const colCount =
   410→    1 + // #
   411→    (f.showDescription ? 1 : 0) +
   412→    (f.showQty ? 1 : 0) +
   413→    (f.showUnitPrice ? 1 : 0) +
   414→    (f.showDiscount ? 1 : 0) +
   415→    (f.showVat ? 1 : 0) +
   416→    1 // Amount
   417→  const values = useMemo(() => buildInvoiceValues(data, settings, docType), [data, settings, docType])
   418→  const title = resolvePlaceholders(config.titlePattern, values)
   419→  const headerNote = config.headerNote ? resolvePlaceholders(config.headerNote, values) : ''
   420→  const footerNote = config.footerNote ? resolvePlaceholders(config.footerNote, values) : ''
   421→
   422→  return (
   423→    <div className="relative bg-white">
   424→      {/* Header */}
   425→      <div className="flex flex-wrap items-start justify-between gap-4">
   426→        <CompanyBlock config={config} settings={settings} />
   427→        <div className="flex flex-col items-end gap-2">
   428→          <h1 className="text-right text-2xl font-extrabold leading-tight" style={{ color: accent }}>
   429→            {title}
   430→          </h1>
   431→          <MetaBox
   432→            rows={[
   433→              { label: 'Status', value: STATUS_LABELS[data.status] ?? data.status },
   434→              { label: 'Date', value: formatDate(data.date) },
   435→              { label: data.dueDate ? dueLabel : '', value: data.dueDate ? formatDate(data.dueDate) : '' },
   436→              { label: 'Linked doc', value: data.linkedRef ?? '' },
   437→              { label: 'Reference', value: data.linkedRef ? data.ref : '' },
   438→            ]}
   439→          />
   440→        </div>
   441→      </div>
   442→
   443→      {headerNote ? <p className="mt-3 text-[12px] italic text-neutral-500">{headerNote}</p> : null}
   444→
   445→      <BillToCard tp={data.thirdParty} accentColor={accent} label={billToLabel} />
   446→
   447→      {/* Lines table */}
   448→      <table className="w-full border-collapse text-[12.5px]">
   449→        <thead>
   450→          <tr className="bg-neutral-100 text-left">
   451→            <th className="border border-neutral-200 px-2.5 py-1.5 font-semibold">#</th>
   452→            {f.showDescription && (
   453→              <th className="border border-neutral-200 px-2.5 py-1.5 font-semibold">Description</th>
   454→            )}
   455→            {f.showQty && <th className="border border-neutral-200 px-2.5 py-1.5 text-right font-semibold">Qty</th>}
   456→            {f.showUnitPrice && (
   457→              <th className="border border-neutral-200 px-2.5 py-1.5 text-right font-semibold">Unit price</th>
   458→            )}
   459→            {f.showDiscount && (
   460→              <th className="border border-neutral-200 px-2.5 py-1.5 text-right font-semibold">Discount %</th>
   461→            )}
   462→            {f.showVat && (
   463→              <th className="border border-neutral-200 px-2.5 py-1.5 text-right font-semibold">VAT %</th>
   464→            )}
   465→            <th className="border border-neutral-200 px-2.5 py-1.5 text-right font-semibold">Amount</th>
   466→          </tr>
   467→        </thead>
   468→        <tbody>
   469→          {data.lines.length === 0 ? (
   470→            <tr>
   471→              <td
   472→                colSpan={colCount}
   473→                className="border border-neutral-200 px-2.5 py-3 text-center text-[11.5px] italic text-neutral-400"
   474→              >
   475→                No line items — totals entered directly
   476→              </td>
   477→            </tr>
   478→          ) : (
   479→            data.lines.map((l, i) => (
   480→            <tr key={`${l.label}-${i}`} className={i % 2 === 1 ? 'bg-neutral-50' : undefined}>
   481→              <td className="border border-neutral-200 px-2.5 py-1.5 text-neutral-500">{i + 1}</td>
   482→              {f.showDescription && (
   483→                <td className="border border-neutral-200 px-2.5 py-1.5">
   484→                  <span className="block font-semibold">{l.label}</span>
   485→                  {l.description ? <span className="block text-[11px] text-neutral-500">{l.description}</span> : null}
   486→                </td>
   487→              )}
   488→              {f.showQty && (
   489→                <td className="border border-neutral-200 px-2.5 py-1.5 text-right tabular-nums">{l.qty}</td>
   490→              )}
   491→              {f.showUnitPrice && (
   492→                <td className="border border-neutral-200 px-2.5 py-1.5 text-right tabular-nums">
   493→                  {formatMoney(l.unitPrice)}
   494→                </td>
   495→              )}
   496→              {f.showDiscount && (
   497→                <td className="border border-neutral-200 px-2.5 py-1.5 text-right tabular-nums">{l.discountPct}</td>
   498→              )}
   499→              {f.showVat && (
   500→                <td className="border border-neutral-200 px-2.5 py-1.5 text-right tabular-nums">{l.vatRate}%</td>
   501→              )}
   502→              <td className="border border-neutral-200 px-2.5 py-1.5 text-right font-medium tabular-nums">
   503→                {formatMoney(computeLineTotal(l))}
   504→              </td>
   505→            </tr>
   506→            ))
   507→          )}
   508→        </tbody>
   509→      </table>
   510→
   511→      {/* Totals */}
   512→      <div className="ml-auto mt-4 w-64 space-y-1 text-[13px]">
   513→        <div className="flex justify-between">
   514→          <span className="text-neutral-500">Subtotal</span>
   515→          <span className="tabular-nums">{formatMoney(data.subtotal)}</span>
   516→        </div>
   517→        {f.showDiscount && data.discountTotal > 0 && (
   518→          <div className="flex justify-between">
   519→            <span className="text-neutral-500">Discount</span>
   520→            <span className="tabular-nums">−{formatMoney(data.discountTotal)}</span>
   521→          </div>
   522→        )}
   523→        {f.showVat && (
   524→          <div className="flex justify-between">
   525→            <span className="text-neutral-500">VAT total</span>
   526→            <span className="tabular-nums">{formatMoney(data.vatTotal)}</span>
   527→          </div>
   528→        )}
   529→        <div className="flex justify-between border-t border-neutral-300 pt-1.5">
   530→          <span className="font-bold">TOTAL</span>
   531→          <span className="text-lg font-extrabold tabular-nums" style={{ color: accent }}>
   532→            {formatMoney(data.total)}
   533→          </span>
   534→        </div>
   535→        {data.paidAmount !== undefined && (
   536→          <div className="flex justify-between">
   537→            <span className="text-neutral-500">Amount paid</span>
   538→            <span className="tabular-nums">{formatMoney(data.paidAmount)}</span>
   539→          </div>
   540→        )}
   541→        {data.paidAmount !== undefined && (
   542→          <div className="flex justify-between font-bold">
   543→            <span>Amount due</span>
   544→            <span className="tabular-nums">{formatMoney(Math.max(0, data.total - data.paidAmount))}</span>
   545→          </div>
   546→        )}
   547→      </div>
   548→
   549→      {/* Notes */}
   550→      {f.showNotes && data.notes ? (
   551→        <p
   552→          className="mt-5 border-l-2 pl-3 text-[12px] italic text-neutral-600"
   553→          style={{ borderColor: accent }}
   554→        >
   555→          {data.notes}
   556→        </p>
   557→      ) : null}
   558→
   559→      <DocFooter config={config} settings={settings} footerNote={footerNote} />
   560→    </div>
   561→  )
   562→}

   565→// Receipt view
   566→// ---------------------------------------------------------------------------
   567→function ReceiptView({
   568→  data,
   569→  config,
   570→  settings,
   571→}: {
   572→  data: ReceiptPrintData
   573→  config: TemplateConfig
   574→  settings: SettingsMap
   575→}) {
   576→  const accent = config.accentColor
   577→  const remaining = Math.max(0, Math.round((data.invoice.total - data.invoice.paidAmount) * 100) / 100)
   578→  const fullyPaid = remaining <= 0.001
   579→  const values = useMemo(() => buildReceiptValues(data, settings), [data, settings])
   580→  const title = resolvePlaceholders(config.titlePattern, values)
   581→  const headerNote = config.headerNote ? resolvePlaceholders(config.headerNote, values) : ''
   582→  const footerNote = config.footerNote ? resolvePlaceholders(config.footerNote, values) : ''
   583→
   584→  const detailRows = [
   585→    { label: 'Payment method', value: PAYMENT_METHOD_LABELS[data.method] ?? data.method },
   586→    { label: 'Payment date', value: formatDate(data.date) },
   587→    { label: 'Reference', value: data.reference ?? data.paymentRef ?? '—' },
   588→    { label: 'For invoice', value: data.invoice.ref },
   589→  ]
   590→
   591→  return (
   592→    <div className="relative bg-white">
   593→      {/* Header */}
   594→      <div className="flex flex-wrap items-start justify-between gap-4">
   595→        <CompanyBlock config={config} settings={settings} />
   596→        <div className="flex flex-col items-end gap-2">
   597→          <h1 className="text-right text-2xl font-extrabold leading-tight" style={{ color: accent }}>
   598→            {title}
   599→          </h1>
   600→          <MetaBox
   601→            rows={[
   602→              { label: 'Invoice', value: data.invoice.ref },
   603→              { label: 'Invoice total', value: formatMoney(data.invoice.total) },
   604→              { label: 'Status', value: fullyPaid ? (STATUS_LABELS.PAID ?? 'Paid') : (STATUS_LABELS.PARTIAL ?? 'Partially paid') },
   605→            ]}
   606→          />
   607→        </div>
   608→      </div>
   609→
   610→      {headerNote ? <p className="mt-3 text-[12px] italic text-neutral-500">{headerNote}</p> : null}
   611→
   612→      <BillToCard tp={data.customer} accentColor={accent} />
   613→
   614→      {/* Receipt panel */}
   615→      <div className="relative my-6 rounded-lg border border-neutral-300 px-6 py-8 text-center">
   616→        {config.fields.showPaidBadge && fullyPaid && <PaidStamp accentColor={accent} />}
   617→        <p className="text-[10px] font-semibold uppercase tracking-widest text-neutral-500">Amount received</p>
   618→        <p className="mt-1 text-4xl font-extrabold tabular-nums" style={{ color: accent }}>
   619→          {formatMoney(data.amount)}
   620→        </p>
   621→        <p className="mt-4 text-[13px]">
   622→          Received from <span className="font-bold">{data.customer.name}</span>{' '}
   623→          <span className="text-neutral-400">({data.customer.code})</span>
   624→        </p>
   625→        <div className="mx-auto mt-4 max-w-xs space-y-1 text-[12.5px]">
   626→          {detailRows.map((r) => (
   627→            <div key={r.label} className="flex justify-between gap-6">
   628→              <span className="text-neutral-500">{r.label}</span>
   629→              <span className="font-medium">{r.value}</span>
   630→            </div>
   631→          ))}
   632→        </div>
   633→        <p className="mt-4 text-[12.5px]">
   634→          Remaining balance on {data.invoice.ref}:{' '}
   635→          <span className={fullyPaid ? 'font-bold text-emerald-700' : 'font-bold text-amber-700'}>
   636→            {formatMoney(remaining)}
   637→          </span>
   638→        </p>
   639→      </div>
   640→
   641→      <DocFooter config={config} settings={settings} footerNote={footerNote} />
   642→    </div>
   643→  )
   644→}

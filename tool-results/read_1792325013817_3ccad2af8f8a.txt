    69→function money(n: number, currency: Currency): string {
    70→  if (currency === 'KHR') {
    71→    return `${new Intl.NumberFormat('en-US', { maximumFractionDigits: 0 }).format(n)} KHR`
    72→  }
    73→  return new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(n)
    74→}
    75→
    76→function parsePresets(raw: string): number[] {
    77→  return raw
    78→    .split(',')
    79→    .map((s) => Number.parseFloat(s.trim()))
    80→    .filter((n) => Number.isFinite(n) && n > 0)
    81→    .slice(0, 8)
    82→}
    83→
    84→function initials(name: string): string {
    85→  return name.split(/\s+/).filter(Boolean).slice(0, 2).map((p) => p[0]?.toUpperCase() ?? '').join('') || '?'
    86→}

    88→function readFileAsDataUrl(file: File): Promise<string> {
    89→  return new Promise((resolve, reject) => {
    90→    const reader = new FileReader()
    91→    reader.onload = () => resolve(String(reader.result))
    92→    reader.onerror = () => reject(reader.error ?? new Error('Could not read the file'))
    93→    reader.readAsDataURL(file)
    94→  })
    95→}
    96→
    97→function khqrFileExtension(dataUrl: string): string {
    98→  const mime = /^data:image\/(\w+)[;,]/.exec(dataUrl)?.[1] ?? 'png'
    99→  return mime === 'jpeg' ? 'jpg' : mime === 'svg+xml' ? 'svg' : mime
   100→}

   102→function downloadDataUrl(dataUrl: string, filename: string) {
   103→  const a = document.createElement('a')
   104→  a.href = dataUrl
   105→  a.download = filename
   106→  document.body.appendChild(a)
   107→  a.click()
   108→  a.remove()
   109→}

   117→export default function CoffeeModule() {
   118→  const user = useAuthStore((s) => s.user)
   119→  const mayEdit = user ? canEdit(user.permissions, 'coffee') : false
   120→  const [tab, setTab] = useState('page')
   121→  const queryClient = useQueryClient()
   122→
   123→  const q = useQuery({ queryKey: ['coffee'], queryFn: () => api<CoffeePayload>('/api/coffee/page') })
   124→
   125→  // Shared config saver — writes the page config and refreshes the cache.
   126→  const [savingPage, setSavingPage] = useState(false)
   127→  async function savePage(patch: Record<string, unknown>, successMsg: string): Promise<boolean> {
   128→    setSavingPage(true)
   129→    try {
   130→      const res = await api<{ page: CoffeePageCfg }>('/api/coffee/page', {
   131→        method: 'PUT',
   132→        body: JSON.stringify(patch),
   133→      })
   134→      queryClient.setQueryData<CoffeePayload>(['coffee'], (old) =>
   135→        old ? { ...old, page: res.page } : old,
   136→      )
   137→      toast.success(successMsg)
   138→      return true
   139→    } catch (err) {
   140→      toast.error(err instanceof Error ? err.message : 'Failed to save the coffee page')
   141→      return false
   142→    } finally {
   143→      setSavingPage(false)
   144→    }
   145→  }
   146→
   147→  // KHQR image upload / removal — saved immediately (server config is the base).
   148→  const [uploading, setUploading] = useState(false)
   149→  async function handleImageFile(file: File) {
   150→    if (!/^image\/(png|jpe?g|webp)$/.test(file.type)) {
   151→      toast.error('Only PNG, JPG or WebP images are supported.')
   152→      return
   153→    }
   154→    if (file.size > MAX_IMAGE_BYTES) {
   155→      toast.error('Image is too large — the maximum size is 2 MB.')
   156→      return
   157→    }
   158→    setUploading(true)
   159→    try {
   160→      const dataUrl = await readFileAsDataUrl(file)
   161→      await savePage({ khqrImage: dataUrl }, 'KHQR image uploaded')
   162→    } catch {
   163→      toast.error('Could not read that file — please try another image.')
   164→    } finally {
   165→      setUploading(false)
   166→    }
   167→  }
   168→  async function removeImage() {
   169→    setUploading(true)
   170→    try {
   171→      await savePage({ khqrImage: null }, 'KHQR image removed')
   172→    } finally {
   173→      setUploading(false)
   174→    }
   175→  }
   176→
   177→  // Support log mutations
   178→  const [deletingId, setDeletingId] = useState<string | null>(null)
   179→  async function deleteSupport(id: string) {
   180→    setDeletingId(id)
   181→    try {
   182→      await api(`/api/coffee/supports/${id}`, { method: 'DELETE' })
   183→      toast.success('Support entry removed')
   184→      await queryClient.invalidateQueries({ queryKey: ['coffee'] })
   185→    } catch (err) {
   186→      toast.error(err instanceof Error ? err.message : 'Failed to delete the support entry')
   187→    } finally {
   188→      setDeletingId(null)
   189→    }
   190→  }
   191→  async function addSupport(body: {
   192→    name: string
   193→    amount: number
   194→    currency: Currency
   195→    message: string | null
   196→    date: string
   197→  }): Promise<boolean> {
   198→    try {
   199→      await api('/api/coffee/supports', { method: 'POST', body: JSON.stringify(body) })
   200→      toast.success('Support recorded — it now shows on your page')
   201→      await queryClient.invalidateQueries({ queryKey: ['coffee'] })
   202→      return true
   203→    } catch (err) {
   204→      toast.error(err instanceof Error ? err.message : 'Failed to record the support')
   205→      return false
   206→    }
   207→  }
   208→
   209→  const page = q.data?.page
   210→  const supporters = q.data?.supporters ?? []
   211→
   212→  return (
   213→    <div className="space-y-6">
   214→      <PageHeader
   215→        title="Buy Me a Coffee"
   216→        description="A friendly supporter page with your KHQR payment QR — scan to send a coffee."
   217→        icon={Coffee}
   218→        actions={
   219→          page ? (
   220→            page.khqrImage ? (
   221→              <Badge variant="outline" className="gap-1.5 border-emerald-600/30 text-emerald-700 dark:text-emerald-400">
   222→                <QrCode className="h-3.5 w-3.5" aria-hidden="true" />
   223→                KHQR ready
   224→              </Badge>
   225→            ) : (
   226→              <Badge variant="outline" className="gap-1.5 border-amber-600/40 text-amber-700 dark:text-amber-400">
   227→                <QrCode className="h-3.5 w-3.5" aria-hidden="true" />
   228→                QR not set yet
   229→              </Badge>
   230→            )
   231→          ) : undefined
   232→        }
   233→      />
   234→
   235→      {q.isLoading && (
   236→        <div className="space-y-4">
   237→          <Skeleton className="h-64 w-full rounded-xl" />
   238→          <div className="grid gap-4 sm:grid-cols-2">
   239→            <Skeleton className="h-40 rounded-xl" />
   240→            <Skeleton className="h-40 rounded-xl" />
   241→          </div>
   242→        </div>
   243→      )}
   244→
   245→      {q.isError && (
   246→        <Alert variant="destructive">
   247→          <Coffee className="h-4 w-4" aria-hidden="true" />
   248→          <AlertTitle>Failed to load the coffee page</AlertTitle>
   249→          <AlertDescription>{(q.error as Error).message}</AlertDescription>
   250→          <Button variant="outline" size="sm" className="mt-3 w-fit" onClick={() => void q.refetch()}>
   251→            Retry
   252→          </Button>
   253→        </Alert>
   254→      )}
   255→
   256→      {q.isSuccess && page && (
   257→        <Tabs value={tab} onValueChange={setTab}>
   258→          <TabsList className="w-full justify-start overflow-x-auto sm:w-auto">
   259→            <TabsTrigger value="page" className="gap-1.5">
   260→              <Coffee className="h-4 w-4" aria-hidden="true" />
   261→              Coffee page
   262→            </TabsTrigger>
   263→            <TabsTrigger value="settings" className="gap-1.5">
   264→              <ImagePlus className="h-4 w-4" aria-hidden="true" />
   265→              Settings
   266→            </TabsTrigger>
   267→            <TabsTrigger value="log" className="gap-1.5">
   268→              <HeartHandshake className="h-4 w-4" aria-hidden="true" />
   269→              Support log
   270→            </TabsTrigger>
   271→          </TabsList>
   272→
   273→          <TabsContent value="page" className="mt-4">
   274→            <CoffeePageView
   275→              key={page.updatedAt ?? 'default'}
   276→              page={page}
   277→              supporters={supporters}
   278→              mayEdit={mayEdit}
   279→              onGoSettings={() => setTab('settings')}
   280→            />
   281→          </TabsContent>
   282→
   283→          <TabsContent value="settings" className="mt-4">
   284→            <CoffeeSettingsTab
   285→              key={page.updatedAt ?? 'default'}
   286→              page={page}
   287→              savingPage={savingPage}
   288→              uploading={uploading}
   289→              mayEdit={mayEdit}
   290→              onImageFile={(f) => void handleImageFile(f)}
   291→              onRemoveImage={() => void removeImage()}
   292→              onSavePatch={(patch, msg) => savePage(patch, msg)}
   293→            />
   294→          </TabsContent>
   295→
   296→          <TabsContent value="log" className="mt-4">
   297→            <SupportLogTab
   298→              page={page}
   299→              supporters={supporters}
   300→              mayEdit={mayEdit}
   301→              onAdd={addSupport}
   302→              onDelete={deleteSupport}
   303→              deletingId={deletingId}
   304→            />
   305→          </TabsContent>
   306→        </Tabs>
   307→      )}
   308→    </div>
   309→  )
   310→}
